// Hardcoded Session ID to bypass workspace CRUD overhead for now
const DEFAULT_SESSION_ID = "default-session";

export default function Home() {
  const { nodes, selectedNodeId, initializeNodes, createNode, setSelectedNodeId } = useNodeStore();
  
  // Connect directly to our Zustand storage store
  const { assets, fetchAssets } = useWorkspaceStore();

  const [activeTab, setActiveTab] = useState<'preview_editor' | 'storage' | 'json'>('preview_editor');
  
  // Transform layer states
  const [pan, setPan] = useState({ x: 150, y: 150 });
  const [zoom, setZoom] = useState(1.0);
  const [isDraggingCanvas, setIsDraggingCanvas] = useState(false);
  const [canvasDragStart, setCanvasDragStart] = useState({ x: 0, y: 0 });

  // 1. Hydrate initial flow nodes if store is completely empty
  useEffect(() => {
    if (Object.keys(nodes).length === 0) {
      initializeNodes(INITIAL_DEMO_GRAPH);
    }
  }, [nodes, initializeNodes]);

  // 2. Fetch the workspace assets from MongoDB Atlas on component mount
  useEffect(() => {
    fetchAssets(DEFAULT_SESSION_ID);
  }, [fetchAssets]);

  // --- MOUSE WORKSPACE INTERACTION HANDLERS ---
  const handleWheel = useCallback((e: React.WheelEvent<HTMLDivElement>) => {
    const zoomIntensity = 0.05;
    const rect = e.currentTarget.getBoundingClientRect();
    const mouseX = e.clientX - rect.left;
    const mouseY = e.clientY - rect.top;

    const wheelValue = e.deltaY < 0 ? 1 : -1;
    const zoomFactor = Math.exp(wheelValue * zoomIntensity);
    const nextZoom = Math.min(Math.max(zoom * zoomFactor, 0.2), 2.5);

    setPan((prevPan) => ({
      x: mouseX - (mouseX - prevPan.x) * (nextZoom / zoom),
      y: mouseY - (mouseY - prevPan.y) * (nextZoom / zoom),
    }));
    setZoom(nextZoom);
  }, [zoom]);

  const handleMouseDown = useCallback((e: React.MouseEvent<HTMLDivElement>) => {
    if ((e.target as HTMLElement).closest('.scenario-node-box')) return;
    setIsDraggingCanvas(true);
    setCanvasDragStart({ x: e.clientX - pan.x, y: e.clientY - pan.y });
  }, [pan]);

  const handleMouseMove = useCallback((e: React.MouseEvent<HTMLDivElement>) => {
    if (!isDraggingCanvas) return;
    setPan({
      x: e.clientX - canvasDragStart.x,
      y: e.clientY - canvasDragStart.y,
    });
  }, [isDraggingCanvas, canvasDragStart]);

  const handleMouseUp = useCallback(() => {
    setIsDraggingCanvas(false);
  }, []);

  const handleExportJSON = () => {
    const exportStructure: any = {};
    Object.keys(nodes).forEach(key => {
      const { x, y, ...rest } = nodes[key];
      exportStructure[key] = rest;
    });
    const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify({ nodes: exportStructure }, null, 2));
    const downloadAnchor = document.createElement('a');
    downloadAnchor.setAttribute("href", dataStr);
    downloadAnchor.setAttribute("download", "linguisquest_story_tree.json");
    document.body.appendChild(downloadAnchor);
    downloadAnchor.click();
    downloadAnchor.remove();
  };

  return (
    <div className="w-screen h-screen flex flex-col bg-[#0b1329] text-slate-100 overflow-hidden font-sans">
      <header className="h-14 bg-[#1e293b] border-b border-slate-700 px-6 flex justify-between items-center shrink-0 shadow-lg z-10 select-none">
        <div>
          <h1 className="text-sm font-bold text-white tracking-wide">Linguis Quest Navigable Flow Grid</h1>
          <p className="text-[10px] text-slate-400 hidden sm:block">
            Scroll Wheel to Zoom. Drag canvas space to Pan. Manage dialogue trees step by step.
          </p>
        </div>
        <div className="flex items-center gap-2">
          <button
            onClick={() => createNode(pan, zoom)}
            className="flex items-center gap-1 bg-indigo-600 hover:bg-indigo-500 text-white text-[11px] font-medium px-2.5 py-1.5 rounded transition-colors"
          >
            <Plus className="w-3.5 h-3.5" /> Spawn Scenario Box
          </button>
          <button
            onClick={handleExportJSON}
            className="flex items-center gap-1 bg-emerald-600 hover:bg-emerald-500 text-white text-[11px] font-medium px-2.5 py-1.5 rounded transition-colors"
          >
            <Download className="w-3.5 h-3.5" /> Export JSON
          </button>
        </div>
      </header>

      <main className="flex-1 flex min-h-0 relative">
        <div className="flex-1 h-full relative">
          <FlowCanvas
            pan={pan}
            zoom={zoom}
            dbAssets={assets} // <-- Wire up Zustand assets directly
            onMouseDown={handleMouseDown} 
            onMouseMove={handleMouseMove}
            onMouseUp={handleMouseUp}
            onWheel={handleWheel}
          />
          
          <MiniRadar
            nodes={nodes}
            zoom={zoom}
            pan={pan}
            onRadarClick={(wx, wy) => {
              setPan({ 
                x: (typeof window !== 'undefined' ? window.innerWidth : 1000) / 2 - wx * zoom, 
                y: (typeof window !== 'undefined' ? window.innerHeight : 600) / 2 - wy * zoom 
              });
            }}
          />
        </div>

        <EditorSidebar
          activeTab={activeTab}
          setActiveTab={setActiveTab}
          sessionId={DEFAULT_SESSION_ID} // <-- Direct sessionId context transmission
        />
      </main>
    </div>
  );
}
