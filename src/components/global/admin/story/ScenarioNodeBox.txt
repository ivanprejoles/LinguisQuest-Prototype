'use client';

import React, { useRef, useState, useEffect } from 'react';
import { GraphNode, useNodeStore } from '@/store/useNodeStore'; // Aligned to your local state node interface
import { Trash2, X, ArrowRight } from 'lucide-react';

// The new cloud asset format representing records stored in MongoDB
export interface Asset {
  fileId: string;
  name: string;
  rowType: string;
  url: string;
  timestamp: number;
}

interface ScenarioNodeBoxProps {
  node: GraphNode;
  isSelected: boolean;
  onSelect: (id: string | null) => void;
  onUpdate: (id: string, fields: Partial<GraphNode>) => void;
  onRemove: (id: string) => void;
  dbAssets: Asset[]; // Updated strictly to map cloud assets

  // 🔗 Drag-and-drop connection wires state passed from FlowCanvas
  isDraggingConnection: boolean;
  onStartDragConnection: (sourceId: string, choiceIdx: number, startX: number, startY: number) => void;
  onCompleteDragConnection: (targetId: string) => void;
}

export function ScenarioNodeBox({
  node,
  isSelected,
  onSelect,
  onUpdate,
  onRemove,
  dbAssets,
  isDraggingConnection,
  onStartDragConnection,
  onCompleteDragConnection
}: ScenarioNodeBoxProps) {
  const boxRef = useRef<HTMLDivElement>(null);
  const [isDragging, setIsDragging] = useState(false);

  // Track continuous click offsets without component-wide local state dependency overhead
  const dragOffset = useRef({ x: 0, y: 0 });

  // Separate runtime storage vectors using the updated cloud database row configurations
  const assetBackgrounds = dbAssets.filter(a => a.rowType === 'background_image');
  const assetCharacters = dbAssets.filter(a => a.rowType === 'character_figure');
  const assetMusic = dbAssets.filter(a => a.rowType === 'background_sound');
  const assetEffects = dbAssets.filter(a => a.rowType === 'sound_effect');

  // Grab all nodes safely at the top level for choice comparison rules
  const allNodes = useNodeStore((state) => state.nodes);

  // Fast MouseDown capture
  const handleHeaderMouseDown = (e: React.MouseEvent) => {
    e.stopPropagation(); 
    onSelect(node.id);
    setIsDragging(true);
    dragOffset.current = {
      x: e.clientX - node.x,
      y: e.clientY - node.y
    };
  };

  // High-Performance Drag Tracking Hook
  useEffect(() => {
    if (!isDragging) return;

    const handleMouseMove = (e: MouseEvent) => {
      // Direct updates bypass local component state rendering queues for immediate 1:1 cursor reactivity
      onUpdate(node.id, {
        x: Math.round(e.clientX - dragOffset.current.x),
        y: Math.round(e.clientY - dragOffset.current.y)
      });
    };

    const handleMouseUp = () => setIsDragging(false);

    window.addEventListener('mousemove', handleMouseMove, { passive: true });
    window.addEventListener('mouseup', handleMouseUp);
    return () => {
      window.removeEventListener('mousemove', handleMouseMove);
      window.removeEventListener('mouseup', handleMouseUp);
    };
  }, [isDragging, node.id, onUpdate]);

  // Triggers wire generation relative to the canvas space coordinates
  const handlePinMouseDown = (e: React.MouseEvent, choiceIndex: number) => {
    e.stopPropagation();
    e.preventDefault();
    
    // Find the current DOM element to get its exact center coordinate relative to the box layout
    const pinEl = e.currentTarget as HTMLElement;
    const canvasContainer = document.getElementById('flow-canvas-container');
    
    if (pinEl && canvasContainer) {
      const pinRect = pinEl.getBoundingClientRect();
      const canvasRect = canvasContainer.getBoundingClientRect();
      
      // Calculate pixel midpoint offset inside the canvas wrapper container frame
      const canvasX = pinRect.left + (pinRect.width / 2) - canvasRect.left;
      const canvasY = pinRect.top + (pinRect.height / 2) - canvasRect.top;
      
      onStartDragConnection(node.id, choiceIndex, canvasX, canvasY);
    }
  };

  return (
    <div
      ref={boxRef}
      onMouseDown={(e) => { e.stopPropagation(); onSelect(node.id); }}
      style={{ position: 'absolute', left: `${node.x}px`, top: `${node.y}px`, transformOrigin: 'top left', pointerEvents: 'auto' }}
      className={`scenario-node-box w-72 bg-[#131c31]/95 border-2 rounded-xl shadow-2xl flex flex-col relative ${
        isSelected ? 'border-indigo-500 shadow-indigo-500/20 ring-1 ring-indigo-500 z-30' : 'border-slate-800 z-20'
      }`}
      data-node-id={node.id}
    >
      {/* 📍 INCOMING PIN DOT (Upgraded to 18x18px) */}
      <div 
        onMouseUp={(e) => { e.stopPropagation(); onCompleteDragConnection(node.id); }}
        className={`absolute -top-[9px] left-1/2 -translate-x-1/2 w-[18px] h-[18px] rounded-full border-2 transition-all cursor-crosshair z-40 ${
          isDraggingConnection
            ? 'bg-emerald-400 border-white scale-110 animate-pulse shadow-lg shadow-emerald-500/50'
            : 'bg-indigo-500 border-[#0c1020]'
        }`} 
        data-pin-target={node.id}
      />

      {/* HEADER BAR */}
      <div onMouseDown={handleHeaderMouseDown} className="p-3 bg-slate-900/60 border-b border-slate-800 rounded-t-xl cursor-move flex items-center justify-between gap-2">
        <div className="flex flex-col min-w-0">
          <input
            type="text"
            value={node.title}
            onMouseDown={(e) => e.stopPropagation()}
            onChange={(e) => onUpdate(node.id, { title: e.target.value })}
            className="bg-transparent font-bold text-xs text-slate-100 focus:outline-none w-40 truncate"
          />
        </div>
        <button onMouseDown={(e) => e.stopPropagation()} onClick={() => onRemove(node.id)} className="p-1 hover:bg-rose-950 rounded text-slate-500 hover:text-rose-400"><Trash2 className="w-3.5 h-3.5" /></button>
      </div>

      {/* STORAGE ASSET SELECTORS PANEL */}
      <div className="p-3 border-b border-slate-800/60 bg-slate-900/20 space-y-2 text-[10px]">
        {/* Background Image Hook */}
        <div className="flex items-center gap-2">
          <span className="w-16 font-bold text-slate-400 flex items-center gap-1">🏞️ Image:</span>
          <select
            value={node.bgImage || ''}
            onMouseDown={(e) => e.stopPropagation()}
            onChange={(e) => onUpdate(node.id, { bgImage: e.target.value })}
            className="flex-1 bg-slate-950 text-slate-200 p-1 border border-slate-800 rounded focus:outline-none focus:border-indigo-500 truncate"
          >
            <option value="">-- No Background --</option>
            {assetBackgrounds.map((a) => <option key={a.fileId} value={a.url}>{a.name}</option>)}
          </select>
        </div>

        {/* Character Figure Hook */}
        <div className="flex items-center gap-2">
          <span className="w-16 font-bold text-slate-400 flex items-center gap-1">🎭 Character:</span>
          <select
            value={node.characterImage || ''}
            onMouseDown={(e) => e.stopPropagation()}
            onChange={(e) => onUpdate(node.id, { characterImage: e.target.value })}
            className="flex-1 bg-slate-950 text-slate-200 p-1 border border-slate-800 rounded focus:outline-none focus:border-indigo-500 truncate"
          >
            <option value="">-- No Character --</option>
            {assetCharacters.map((a) => <option key={a.fileId} value={a.url}>{a.name}</option>)}
          </select>
        </div>

        {/* 🎵 Ambient Audio Track Assignment */}
        <div className="flex items-center gap-2">
          <span className="w-16 font-bold text-slate-400 flex items-center gap-1">🎵 Ambient:</span>
          <select
            value={node.bgMusic || ''}
            onMouseDown={(e) => e.stopPropagation()}
            onChange={(e) => onUpdate(node.id, { bgMusic: e.target.value })}
            className="flex-1 bg-slate-950 text-slate-200 p-1 border border-slate-800 rounded focus:outline-none focus:border-indigo-500 truncate"
          >
            <option value="">-- No Music Track --</option>
            {assetMusic.map((a) => <option key={a.fileId} value={a.url}>{a.name}</option>)}
          </select>
        </div>

        {/* 💥 Audio Effects Trigger Selector */}
        <div className="flex items-center gap-2">
          <span className="w-16 font-bold text-slate-400 flex items-center gap-1">💥 Effect:</span>
          <select
            value={node.sfx || ''}
            onMouseDown={(e) => e.stopPropagation()}
            onChange={(e) => onUpdate(node.id, { sfx: e.target.value })}
            className="flex-1 bg-slate-950 text-slate-200 p-1 border border-slate-800 rounded focus:outline-none focus:border-indigo-500 truncate"
          >
            <option value="">-- No Sound FX --</option>
            {assetEffects.map((a) => <option key={a.fileId} value={a.url}>{a.name}</option>)}
          </select>
        </div>
      </div>

      {/* DIALOGUE LINES VAULT */}
      <div className="p-3 flex flex-col gap-1.5">
        <label className="text-[9px] font-bold uppercase text-slate-400 flex justify-between items-center">
          <span>Dialogue Lines ({node.dialogue.length})</span>
          <button onMouseDown={(e) => e.stopPropagation()} onClick={() => onUpdate(node.id, { dialogue: [...node.dialogue, ""] })} className="text-indigo-400 font-bold text-[10px]">+ Add</button>
        </label>
        <div className="space-y-1.5 max-h-24 overflow-y-auto pr-1">
          {node.dialogue.map((line, idx) => (
            <div key={idx} className="flex gap-1 items-start">
              <textarea
                value={line}
                rows={1}
                onMouseDown={(e) => e.stopPropagation()}
                onChange={(e) => {
                  const arr = [...node.dialogue];
                  arr[idx] = e.target.value;
                  onUpdate(node.id, { dialogue: arr });
                }}
                className="w-full bg-slate-950 border border-slate-800 p-1 rounded text-[11px] text-slate-300 resize-none focus:outline-none focus:border-indigo-500"
              />
            </div>
          ))}
        </div>
      </div>

      {/* PATH CONNECTION ROUTES SECTION */}
      <div className="mt-auto p-3 bg-slate-900/30 border-t border-slate-800/60 rounded-b-xl flex flex-col gap-2 relative">
        <div className="flex justify-between items-center">
          <span className="text-[9px] font-bold uppercase text-slate-400">Decision Paths</span>
          <button
            onMouseDown={(e) => e.stopPropagation()}
            onClick={() => onUpdate(node.id, { choices: [...node.choices, { text: 'Next Path Step', targetNodeId: '' }] })}
            className="text-[9px] bg-slate-800 hover:bg-slate-700 text-slate-300 font-bold px-1.5 py-0.5 rounded"
          >
            + Path
          </button>
        </div>

        <div className="space-y-1.5">
          {node.choices.map((choice, idx) => (
            <div key={idx} className="bg-slate-950/80 border border-slate-800 p-1.5 rounded-lg flex flex-col gap-1 relative group/path">
              <div className="flex gap-1 items-center">
                <input
                  type="text"
                  value={choice.text}
                  onMouseDown={(e) => e.stopPropagation()}
                  onChange={(e) => {
                    const paths = [...node.choices];
                    paths[idx].text = e.target.value;
                    onUpdate(node.id, { choices: paths });
                  }}
                  placeholder="Choice option description text..."
                  className="flex-1 bg-transparent text-[11px] text-slate-200 focus:outline-none border-b border-transparent focus:border-indigo-500 py-0.5"
                />
                <button
                  onMouseDown={(e) => e.stopPropagation()}
                  onClick={() => onUpdate(node.id, { choices: node.choices.filter((_, i) => i !== idx) })}
                  className="text-slate-500 hover:text-rose-400 ml-1"
                >
                  <X className="w-3 h-3" />
                </button>
              </div>

              {/* Flex row containing connection feedback logic */}
              <div className="flex gap-1.5 items-center justify-between">
                <div className="flex items-center gap-1 text-[10px] text-slate-400 font-medium">
                  <ArrowRight className="w-3 h-3 text-indigo-400" /> Target Node ID:
                </div>
                <input
                  type="text"
                  placeholder="e.g. node_2"
                  value={choice.targetNodeId}
                  onMouseDown={(e) => e.stopPropagation()}
                  onChange={(e) => {
                    const paths = [...node.choices];
                    paths[idx].targetNodeId = e.target.value;
                    onUpdate(node.id, { choices: paths });
                  }}
                  className="w-24 bg-slate-900 border border-slate-800 focus:border-indigo-500 rounded text-center font-mono text-[10px] py-0.5 text-amber-400 focus:outline-none mr-3"
                />
              </div>

              {/* 📍 OUTGOING PIN DOT (Upgraded: 18x18px, centered on Y-axis, positioned flush on the card outline border edge) */}
              <div 
                onMouseDown={(e) => handlePinMouseDown(e, idx)}
                className="absolute -right-5.5 top-1/2 -translate-y-1/2 w-[18px] h-[18px] bg-indigo-400 hover:bg-indigo-300 rounded-full border-2 border-[#131c31] cursor-crosshair z-40 shadow-md transition-transform hover:scale-110"
                title="Drag wire connection link from here"
                data-pin-source={`${node.id}-${idx}`}
              />
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}
