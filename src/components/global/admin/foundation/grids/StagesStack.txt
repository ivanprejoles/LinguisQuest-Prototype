"use client";

import React from "react";
import { Button } from "@/components/ui/button";
import { GripVertical, Trash2, HelpCircle, ArrowRightLeft, AlignLeft, Layers } from "lucide-react";

export function StagesStack({ stages, onDelete, onDragStart, onDrop }: any) {
  const getTypeBadge = (type: string) => {
    switch (type) {
      case "multipleChoice": return { label: "Multiple Choice", icon: HelpCircle, color: "bg-blue-100 text-blue-700 border-blue-200" };
      case "matching": return { label: "Matching Pairs", icon: ArrowRightLeft, color: "bg-purple-100 text-purple-700 border-purple-200" };
      case "sentenceBuilder": return { label: "Sentence Builder", icon: AlignLeft, color: "bg-emerald-100 text-emerald-700 border-emerald-200" };
      default: return { label: type, icon: Layers, color: "bg-slate-100 text-slate-700 border-slate-200" };
    }
  };

  return (
    <div className="max-w-2xl mx-auto space-y-3">
      {stages.map((stg: any, index: number) => {
        const badge = getTypeBadge(stg.type);
        const Icon = badge.icon;
        return (
          <div
            key={stg.id}
            draggable
            onDragStart={() => onDragStart(index)}
            onDragOver={(e) => e.preventDefault()}
            onDrop={() => onDrop(index)}
            className="group relative bg-white border-2 border-slate-200 hover:border-green-500 rounded-2xl p-4 shadow-[0_4px_0_0_#e2e8f0] transition-all duration-150 flex items-center justify-between gap-4"
          >
            <div className="flex items-center gap-3">
              <GripVertical className="w-5 h-5 text-slate-300 cursor-grab opacity-50 group-hover:opacity-100" />
              <div className="w-10 h-10 rounded-xl bg-slate-900 text-white font-black text-base flex items-center justify-center shadow-sm">
                {index + 1}
              </div>
              <div className="space-y-1">
                <div className="flex items-center gap-2">
                  <span className={`text-[10px] font-black uppercase tracking-wider px-2 py-0.5 rounded-full border flex items-center gap-1 ${badge.color}`}>
                    <Icon className="w-3 h-3" />
                    {badge.label}
                  </span>
                  {stg.audioUrl && <span className="text-[10px] font-bold bg-red-100 text-red-600 px-1.5 py-0.5 rounded">🎙️ Audio</span>}
                </div>
                <p className="font-extrabold text-slate-800 text-sm leading-none">{stg.question}</p>
              </div>
            </div>
            <Button size="icon" variant="ghost" className="h-8 w-8 text-slate-300 hover:text-red-500 opacity-0 group-hover:opacity-100 transition-opacity" onClick={(e) => onDelete(stg.id, e)}>
              <Trash2 className="w-4 h-4" />
            </Button>
          </div>
        );
      })}
    </div>
  );
}
